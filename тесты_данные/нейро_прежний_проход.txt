// Прежние функции прохода пакета модуля «нейро» (библиотека/нейро.dan до потока 24Б): сдвиги, поворот,
// запись в кэш, внимание и SwiGLU считались по токенам. Тест test_neyro.ТестыПрежнегоПрохода собирает из
// нынешнего модуля прежний, заменяя этими функциями одноимённые, и сверяет логиты побитно.

/// Значения кэша дроб16 голов ключей слоя л для позиций 0..длина - 1 в дроб32 подряд по головам ключей
/// (блок длина * ширина головы на голову), на всех ядрах; при кэше дроб32 - пустой массив (значения
/// берутся из кэша без копии). Пакет токенов (Модель.проход) переводит их один раз на слой, а не на
/// каждый токен: числа те же
функ Модель.значения_слоя(н: Сеанс, л: цел, длина: цел) -> массив<дроб32> {
    пусть нкв = сам.голов_кв
    пусть блок = длина * сам.ширина_головы
    пусть первый = л * нкв
    перем зн = массив(дроб32, 0)
    если не сам.вн.точный_кэш {
        зн = массив(дроб32, нкв * блок)
        параллельно для гкв в 0..нкв - 1 {
            зн[гкв * блок..гкв * блок + блок - 1] = в_дроб32(н.кэш16.значения[первый + гкв][0..блок - 1])
        }
    }
    вернуть зн
}

/// Повернуть векторы голов (RoPE) на позицию: кос и син - по паре измерений
функ повернуть(х: массив<дроб32>, голов: цел, шг: цел, изм: цел, вид: цел, кос: [дроб], син: [дроб]) -> массив<дроб32> {
    перем р = х
    пусть пол = частное(изм, 2)
    для г в 0..голов - 1 {
        пусть о = г * шг
        для j в 0..пол - 1 {
            перем а = о + 2 * j
            перем б = а + 1
            если вид == 2 {
                а = о + j
                б = о + j + пол
            }
            пусть х0 = х[а]
            пусть х1 = х[б]
            р[а] = х0 * кос[j] - х1 * син[j]
            р[б] = х0 * син[j] + х1 * кос[j]
        }
    }
    вернуть р
}

функ сложить_сдвиг(х: массив<дроб32>, сдвиг: массив<дроб32>) -> массив<дроб32> {
    если сдвиг.длина == 0 { вернуть х }
    вернуть прибавить_кратное(х, 1.0, сдвиг)
}

/// Внимание одного токена в позиции поз по кэшу слоя л: результат - голов * ширина головы чисел.
/// Головы считаются в «параллельно для», каждая - своим блоком результата; числа каждой головы те же,
/// что при последовательном счёте (ядра внутри шага работают на одном потоке и дают те же биты).
функ Модель.внимание(н: Сеанс, л: цел, поз: цел, q: массив<дроб32>) -> массив<дроб32> {
    вернуть сам.внимание_по(н, л, поз, q, сам.значения_слоя(н, л, поз + 1), поз + 1)
}

/// То же по значениям слоя зн (Модель.значения_слоя для длины длина_зн не меньше поз + 1)
функ Модель.внимание_по(н: Сеанс, л: цел, поз: цел, q: массив<дроб32>, зн: массив<дроб32>, длина_зн: цел) -> массив<дроб32> {
    пусть шг = сам.ширина_головы
    пусть нкв = сам.голов_кв
    пусть голов = сам.голов
    пусть группа = частное(голов, нкв)
    пусть масштаб = 1.0 / корень(дроб(шг))
    пусть длина = поз + 1
    пусть точный = сам.вн.точный_кэш
    пусть первый = л * нкв
    // ключи и значения голов ключей слоя до позиции поз - части кэша (или зн), которые шаги передают в ядра без копии
    пусть блок = длина * шг
    пусть блок_зн = длина_зн * шг
    перем внимание = массив(дроб32, голов * шг)
    параллельно для г в 0..голов - 1 {
        пусть гкв = частное(г, группа)
        перем оценки = массив(дроб32, 0)
        если точный {
            оценки = матвек(н.ключи[первый + гкв][0..блок - 1], q[г * шг..г * шг + шг - 1])
        } иначе {
            оценки = матвек(н.кэш16.ключи[первый + гкв][0..блок - 1], q[г * шг..г * шг + шг - 1])
        }
        для т в 0..длина - 1 { оценки[т] = оценки[т] * масштаб }
        пусть вер = софтмакс(оценки)
        перем о = массив(дроб32, 0)
        если точный {
            о = матмат(вер, н.значения[первый + гкв][0..блок - 1], длина)
        } иначе {
            о = матмат(вер, зн[гкв * блок_зн..гкв * блок_зн + блок - 1], длина)
        }
        внимание[г * шг..г * шг + шг - 1] = о
    }
    вернуть внимание
}

/// Проход пакета токенов (номера проверены, все помещаются в контекст): кэш пополняется, логиты -
/// после последнего токена. Матрицы весов умножаются на векторы всех токенов пакета сразу
/// (Вес.умножить_пакет), остальное считается по токенам так же, как при подаче по одному, поэтому
/// числа те же. все_логиты = истина - логиты после каждого токена пакета (в логиты_пакета; выход
/// умножается на все векторы сразу, каждое число то же, что у Вес.умножить).
функ Модель.проход(с: Сеанс, иды: [цел], все_логиты: лог) -> Сеанс {
    перем н = с
    н.ошибка = ""
    н.логиты_пакета = []
    пусть п0 = н.токены.длина
    пусть т_ = иды.длина
    перем хи: [массив<дроб32>] = []
    для ид в иды {
        если пусть х0 = сам.встраивание_токена(ид) {
            хи.добавить(х0)
        } иначе ошибка о {
            // не «вернуть с»: с после «перем н = с» не используется, поэтому кэш переходит к н без копии
            н.ошибка = о
            вернуть н
        }
    }
    пусть шг = сам.ширина_головы
    пусть нкв = сам.голов_кв
    перем уги: [[[дроб]]] = []
    для т в 0..т_ - 1 { уги.добавить(углы(сам, п0 + т)) }
    для л в 0..сам.слоёв - 1 {
        пусть сл = сам.слои[л]
        перем хн: [массив<дроб32>] = []
        для х в хи { хн.добавить(нормировать_скв(х, сл.норма_внимания, сам.эпсилон)) }
        пусть запросы = сл.запрос.умножить_пакет(хн, сам.вн.кусок)
        пусть ключи_ = сл.ключ.умножить_пакет(хн, сам.вн.кусок)
        пусть значения_ = сл.значение.умножить_пакет(хн, сам.вн.кусок)
        перем повёрнутые: [массив<дроб32>] = []
        для т в 0..т_ - 1 {
            пусть уг = уги[т]
            повёрнутые.добавить(повернуть(сложить_сдвиг(запросы[т], сл.сдвиг_запроса), сам.голов, шг, сам.измерений_rope,
                сам.вид_rope, уг[0], уг[1]))
            пусть k = повернуть(сложить_сдвиг(ключи_[т], сл.сдвиг_ключа), нкв, шг, сам.измерений_rope,
                сам.вид_rope, уг[0], уг[1])
            пусть v = сложить_сдвиг(значения_[т], сл.сдвиг_значения)
            пусть поз = п0 + т
            для г в 0..нкв - 1 {
                пусть номер = л * нкв + г
                если сам.вн.точный_кэш {
                    н.ключи[номер][поз * шг..поз * шг + шг - 1] = k[г * шг..г * шг + шг - 1]
                    н.значения[номер][поз * шг..поз * шг + шг - 1] = v[г * шг..г * шг + шг - 1]
                } иначе {
                    для j в 0..шг - 1 {
                        н.кэш16.ключи[номер][поз * шг + j] = в_половинный(k[г * шг + j])
                        н.кэш16.значения[номер][поз * шг + j] = в_половинный(v[г * шг + j])
                    }
                }
            }
        }
        перем внимания: [массив<дроб32>] = []
        // значения слоя в дроб32 - один раз на пакет
        пусть зн_слоя = сам.значения_слоя(н, л, п0 + т_)
        для т в 0..т_ - 1 { внимания.добавить(сам.внимание_по(н, л, п0 + т, повёрнутые[т], зн_слоя, п0 + т_)) }
        пусть выходы = сл.выход.умножить_пакет(внимания, сам.вн.кусок)
        перем хн2: [массив<дроб32>] = []
        для т в 0..т_ - 1 {
            хи[т] = прибавить_кратное(хи[т], 1.0, выходы[т])
            хн2.добавить(нормировать_скв(хи[т], сл.норма_ffn, сам.эпсилон))
        }
        пусть затворы = сл.затвор.умножить_пакет(хн2, сам.вн.кусок)
        пусть вверх = сл.вверх.умножить_пакет(хн2, сам.вн.кусок)
        перем акт: [массив<дроб32>] = []
        для т в 0..т_ - 1 {
            пусть затв = затворы[т]
            перем вв = вверх[т]
            // SwiGLU на всех ядрах: каждый шаг пишет свой элемент
            параллельно для j в 0..вв.длина - 1 {
                пусть з = затв[j]
                вв[j] = з / (1.0 + экспонента(-з)) * вв[j]
            }
            акт.добавить(вв)
        }
        пусть вниз = сл.вниз.умножить_пакет(акт, сам.вн.кусок)
        для т в 0..т_ - 1 { хи[т] = прибавить_кратное(хи[т], 1.0, вниз[т]) }
    }
    если все_логиты {
        перем хв: [массив<дроб32>] = []
        для х в хи { хв.добавить(нормировать_скв(х, сам.норма_выхода, сам.эпсилон)) }
        н.логиты_пакета = сам.выход.умножить_пакет(хв, сам.вн.кусок)
        н.логиты = н.логиты_пакета[т_ - 1]
    } иначе {
        пусть хк = нормировать_скв(хи[т_ - 1], сам.норма_выхода, сам.эпсилон)
        н.логиты = сам.выход.умножить(хк)
    }
    для ид в иды { н.токены.добавить(ид) }
    вернуть н
}
